# 实体层接口（Entity） — 新增

> 引入目的：消除各层之间裸用 `Dictionary` 的隐患，提供类型安全和代码补全。

### 2.1 DeckEntity

```gdscript
class_name DeckEntity

var id: int = 0
var name: String = ""
var parent_id: int = 0      # 0 表示根级
var sort_order: int = 0
var is_archived: bool = false
var created_at: int = 0     # Unix timestamp
var updated_at: int = 0

# 辅助方法
func to_dict() -> Dictionary
func from_dict(d: Dictionary) -> void
func is_root() -> bool
func get_path_separator() -> String   # 返回 "::"
func get_full_path(parent_name: String = "") -> String # 返回完整路径字符串
static func now_timestamp() -> int
```

### 2.2 NoteEntity

```gdscript
class_name NoteEntity

var id: int = 0

var fields_data: Dictionary = {}    # 字段名到字段值，替代纯 JSON 字符串

func fields_from_json(json_str: String) -> void
func get_default_deck_id() -> int    # 预留获取默认牌组ID
static func now_timestamp() -> int
```

### 2.3 CardEntity

```gdscript
class_name CardEntity

# 队列常量
const QUEUE_NEW: int        = 0
const QUEUE_LEARNING: int   = 1
const QUEUE_REVIEW: int     = 2
const QUEUE_SUSPENDED: int  = -1
const QUEUE_BURIED: int     = -2

# 评分常量
const RATING_AGAIN: int = 1
const RATING_HARD: int  = 2
const RATING_GOOD: int  = 3
const RATING_EASY: int  = 4

var id: int = 0
var note_id: int = 0
var deck_id: int = 0
var template_order: int = 0         # 一个 Note 可生成多张 Card，此为模板序号

var queue: int = QUEUE_NEW
var due: int = 0                    # new=position, review=day_index, learn=timestamp
var reps: int = 0                   # 总复习次数
var lapses: int = 0                 # 遗忘/失败次数
var last_review_time: int = 0       # 上次复习 Unix timestamp
var last_rating: int = 0            
var last_time_taken: int = 0        # 上次答题耗时（毫秒）
var review_history_json: String = "[]"

# FSRS 参数（预留算法扩展）
var stability: float = 0.0
var difficulty: float = 0.0

# 辅助方法
func to_dict() -> Dictionary
func from_dict(d: Dictionary) -> void
func is_due(now_day_index: int) -> bool
func is_new() -> bool
func is_learning() -> bool
func is_review() -> bool
func is_suspended() -> bool
func is_buried() -> bool
func append_review_history(rating: int, time_taken_ms: int, reviewed_at: int = 0) -> void
func get_review_history() -> Array
func get_queue_name() -> String
func get_last_rating_name() -> String
static func now_timestamp() -> int
static func day_index_from_timestamp(unix_ts: int) -> int
static func today_day_index() -> int
```
