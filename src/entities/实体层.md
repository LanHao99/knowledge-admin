# 实体层接口（Entity） —— 新增

> 引入目的：消除各层之间裸传 `Dictionary` 的隐患，提供类型安全和代码补全。

### 2.1 DeckEntity

```gdscript
class_name DeckEntity

var id: int = 0
var name: String = ""
var parent_id: int = 0      # 0 表示根级
var sort_order: int = 0
var is_archived: bool = false
var created_at: int = 0     # Unix timestamp
var updated_at: int = 0

# 辅助方法
func to_dict() -> Dictionary
func from_dict(d: Dictionary) -> void
func is_root() -> bool
func get_path_separator() -> String   # 返回 "::" 或 "/"
```

### 2.2 NoteEntity

```gdscript
class_name NoteEntity

var id: int = 0
var note_type_id: int = 0
var fields_data: Dictionary = {}    # 字段名 → 字段值，替代裸 JSON 字符串
var tags: Array[String] = []        # 预留标签数组
var created_at: int = 0

# 辅助方法
func to_dict() -> Dictionary
func from_dict(d: Dictionary) -> void
func get_field(field_name: String) -> String
func set_field(field_name: String, value: String) -> void
func fields_to_json() -> String      # 写入数据库前序列化
func fields_from_json(json_str: String) -> void
```

### 2.3 CardEntity

```gdscript
class_name CardEntity

var id: int = 0
var note_id: int = 0
var deck_id: int = 0
var template_order: int = 0         # 一个 Note 可生成多张 Card，此为模板序号

# 队列状态（对齐 Anki 语义）
var queue: int = 0                  # 0=new, 1=learning, 2=review, -1=suspended, -2=buried
var due: int = 0                    # new=position, review=day_index, learn=timestamp
var reps: int = 0                   # 总复习次数
var lapses: int = 0                 # 遗忘/失败次数
var last_review_time: int = 0       # 上次复习 Unix timestamp
var last_rating: int = 0            # 1=Again, 2=Hard, 3=Good, 4=Easy
var last_time_taken: int = 0        # 上次答题耗时（毫秒）
var review_history_json: String = "[]"

# FSRS 参数（预留算法扩展）
var stability: float = 0.0
var difficulty: float = 0.0

# 辅助方法
func to_dict() -> Dictionary
func from_dict(d: Dictionary) -> void
func is_due(now_day_index: int) -> bool
func is_new() -> bool
func is_learning() -> bool
func is_review() -> bool
func is_suspended() -> bool
```
